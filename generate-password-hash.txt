Run this command in a PowerShell terminal from the project folder. Do not paste it into .env.local.

$secure = Read-Host "Admin password" -AsSecureString
$plain = [System.Net.NetworkCredential]::new("", $secure).Password
$previousEncoding = $OutputEncoding

try {
  $OutputEncoding = [System.Text.UTF8Encoding]::new($false)

  $hash = $plain | node -e "let value=''; process.stdin.setEncoding('utf8'); process.stdin.on('data', chunk => value += chunk); process.stdin.on('end', async () => console.log(await require('bcryptjs').hash(value.replace(/\r?\n$/, ''), 12)))"

  if ($LASTEXITCODE -ne 0) {
    throw "Hash generation failed. Make sure you run this from the project folder, where bcryptjs is installed."
  }

  $hash = $hash.Trim()
  if ($hash -notmatch '^\$2[aby]\$\d{2}\$[./A-Za-z0-9]{53}$') {
    throw "The generated output doesn't look like a bcrypt hash."
  }

  $hash
}
finally {
  $OutputEncoding = $previousEncoding
  Remove-Variable secure, plain -ErrorAction SilentlyContinue
}
